实际代码：历史编码与逐小时滚动预测

历史海况先编码为循环状态，下一小时风与上一步预测共同驱动解码；未来真实波高只参与监督。以下为类方法节选，需结合完整模块中的定义使用。

def forward(self, history, future_wind):
        batch, _, _, rows, columns = history.shape
        shape = (batch, self.encoder.hidden_channels, rows, columns)
        state = (history.new_zeros(shape), history.new_zeros(shape))
        for frame in history.unbind(1):
            hidden, memory = self.encoder(frame, state)
            state = hidden*self.wet, memory*self.wet
        previous = history[:, -1, :1]*self.hs_scale
        quality = history[:, -1, 4:5]
        wet = self.wet.expand(batch, -1, -1, -1)
        forecasts = []
        for wind in future_wind.unbind(1):
            inputs = torch.cat([previous/self.hs_scale, wind, wet, quality], 1)
            hidden, memory = self.decoder(inputs, state)
            state = hidden*self.wet, memory*self.wet
            previous = torch.relu(previous+self.output(state[0])*self.hs_scale)*self.wet
            forecasts.append(previous[:, 0])
            quality = wet
        return torch.stack(forecasts, 1)
实际代码：ConvLSTM门控与状态更新
class ConvLSTMCell(nn.Module):
    def __init__(self, input_channels, hidden_channels):
        super().__init__()
        self.hidden_channels = hidden_channels
        self.gates = nn.Conv2d(input_channels+hidden_channels, 4*hidden_channels, 3, padding=1)

    def forward(self, inputs, state):
        hidden, memory = state
        incoming, forget, outgoing, candidate = self.gates(torch.cat([inputs, hidden], 1)).chunk(4, 1)
        memory = torch.sigmoid(forget)*memory + torch.sigmoid(incoming)*torch.tanh(candidate)
        hidden = torch.sigmoid(outgoing)*torch.tanh(memory)
        return hidden, memory
实际代码：有效海点的Huber损失
def masked_huber(prediction, target, mask, delta):
    error = prediction-target
    absolute = error.abs()
    point_loss = torch.where(absolute <= delta, 0.5*error.square(), delta*(absolute-0.5*delta))
    counts = mask.sum(dim=(0, 2, 3))
    usable = counts > 0
    if not usable.any():
        raise ValueError('Batch has no valid supervision')
    hourly_loss = (point_loss*mask).sum(dim=(0, 2, 3))/counts.clamp_min(1)
    return hourly_loss[usable].mean()

optimizer = torch.optim.Adam(model.parameters(), lr=config['learning_rate'])

for horizon, epochs in config['training_stages']:
        for _ in range(epochs):
            epoch += 1
            model.train()
            train_loss = []
            for history, wind, target, valid, _ in train_loader:
                optimizer.zero_grad(set_to_none=True)
                prediction = model(history, wind[:, :horizon])
                loss = masked_huber(prediction, target[:, :horizon], valid[:, :horizon], config['huber_delta_m'])
                loss.backward()
                nn.utils.clip_grad_norm_(model.parameters(), 1.0, error_if_nonfinite=True)
                optimizer.step()
                train_loss.append(float(loss.detach()))
            prediction, target, valid, _ = predict(model, validation_loader)
            score = metrics(prediction, target, valid)['mean_lead_rmse_m']
            if score < best_score:
                best_score = score
                best_state = copy.deepcopy(model.state_dict())
                torch.save({'model': best_state, 'config': config,
                            'normalization': audit['normalization_training_only'],
                            'epoch': epoch}, output/'best_model.pt')
            row = {'epoch': epoch, 'training_horizon': horizon, 'loss': float(np.mean(train_loss)),
                   'validation_mean_lead_rmse_m': score, 'elapsed_seconds': time.perf_counter()-started}
            history_log.append(row)

05 后续训练与验证建议

  1. 先落实业务输入来源

    确认未来24小时预报风，以及起报时可及时取得的过去24小时波高资料。

  2. 固定现有模型，替换预报风回放

    用当时已发布的GFS风替换ERA5未来风，保持其他输入一致，逐小时比较误差变化。

  3. 再做针对性优化与独立检验

    根据验证期问题调整模型；用保留时段、目标区独立测波资料检查效果，同时核验185.2米输出网格的适用性。

结果边界：参考答案为WNP产品，未来风为ERA5事后再分析，尚非现场准确率。0.1°资料插值到185.2米，不等于已获得该尺度的湾内预测能力；本页不作为招标达标证明。
